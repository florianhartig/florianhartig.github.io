{{- /* Word-sized inline bar chart ("sparkline") for use inside a sentence. Args (dict): years, values (parallel lists of ints), label (what is counted), asof (fetch date, default build date). The year of asof is drawn lighter (incomplete). Values are in per-bar tooltips and in the aria-label. */ -}} {{- $years := .years }} {{- $vals := .values }} {{- if gt (len $years) 0 }} {{- $max := 0 }} {{- range $vals }}{{ if gt . $max }}{{ $max = . }}{{ end }}{{ end }} {{- $now := cond (isset . "asof") (time.AsTime (.asof | default now)).Year now.Year }} {{- $parts := slice }} {{- range $i, $y := $years }}{{ $parts = $parts | append (printf "%d%s: %s" $y (cond (eq $y $now) " (to date)" "") (lang.FormatNumber 0 (index $vals $i))) }}{{ end -}} {{- range $i, $y := $years }} {{- $v := index $vals $i }} {{- $h := 0.0 }}{{ if gt $max 0 }}{{ $h = div (mul 100.0 $v) $max }}{{ end -}} {{- end -}} {{- end -}}
